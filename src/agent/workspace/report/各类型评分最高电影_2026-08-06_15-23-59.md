## 3. 生成 SQL

```sql
WITH ranked AS (
  SELECT
    g.genre,
    t.primary_title AS movie_title,
    r.average_rating,
    ROW_NUMBER() OVER (PARTITION BY g.genre ORDER BY r.average_rating DESC) AS rn
  FROM titles t
  JOIN title_genres g ON t.tconst = g.tconst
  JOIN title_ratings r ON t.tconst = r.tconst
  WHERE t.title_type = 'movie'
    AND r.num_votes > 10000
    AND r.average_rating IS NOT NULL
)
SELECT genre, movie_title, average_rating
FROM ranked
WHERE rn = 1
ORDER BY average_rating DESC;
```
