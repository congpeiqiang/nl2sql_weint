    78	def _sanitize_echarts_result(result: Any) -&gt; Any:
    79	    """将 ECharts 生成的图表结果转为可渲染内容。
    80	
    81	    支持两种输出：
    82	    - SVG 字符串（outputType="svg"）：转为内嵌 iframe，与 Semiotic 引擎一致，可正常渲染
    83	    - PNG 文件路径（outputType="png"）：转为 &lt;img&gt; 标签，并自动复制到工作区 report 目录
    84	    """
    85	    import logging
    86	    _log = logging.getLogger(__name__)
    87	    _log.warning(f"[ECHARTS] ENTER: type={type(result).__name__}, preview={str(result)[:200]}")
    88	
    89	    # 处理 MCP 标准响应格式：{content: [{type:"text", text:"&lt;svg&gt;..."}]} 或 {content:[{type:"image",...}]}
    90	    if isinstance(result, dict) and isinstance(result.get("content"), list):
    91	        for item in result["content"]:
    92	            if not isinstance(item, dict):
    93	                continue
    94	            if item.get("type") == "text" and "&lt;svg" in str(item.get("text", "")):
    95	                r = _svg_to_data_url(item["text"])
    96	                _log.info(f"[ECHARTS] MCP content text SVG → iframe[:100]={r[:100]}")
    97	                return r
    98	            if item.get("type") == "image":
    99	                data = item.get("data", "")
   100	                mime = item.get("mimeType", "image/png")
   101	                if data:
   102	                    return f'&lt;img src="data:{mime};base64,{data}" style="max-width:100%;border-radius:8px"/&gt;'
   103	        # 若 content 中无 SVG/图片，尝试拼接所有 text 内容
   104	        texts = [str(i.get("text", "")) for i in result["content"] if isinstance(i, dict) and i.get("type") == "text"]
   105	        joined = "".join(texts)
   106	        if "&lt;svg" in joined:
   107	            return _svg_to_data_url(joined)
   108	
   109	    # ECharts MCP 返回的是文件路径字符串（如 C:\\Users\\xxx\\Downloads\\uuid.png）
   110	    if isinstance(result, str):
   111	        _log.info(f"[ECHARTS] str result: {result[:200]}")
   112	        # 检查是否为 SVG 字符串（outputType="svg" 时返回）
   113	        if "&lt;svg" in result:
   114	            r = _svg_to_data_url(result)
   115	            _log.info(f"[ECHARTS] str SVG → iframe[:100]={r[:100]}")
   116	            return r
   117	        # 检查是否为文件路径（包含 .png 等图片扩展名）
   118	        if result.lower().endswith(('.png', '.jpg', '.jpeg', '.gif', '.webp')):
   119	            # 自动复制到工作区 report 目录
   120	            moved = _move_echarts_image_to_workspace(result)
   121	            return f'&lt;img src="file:///{moved.replace(chr(92), "/")}" style="max-width:100%;border-radius:8px"/&gt;'
   122	        # 检查是否为 data URL 或 http URL
   123	        if result.startswith('data:image'):
   124	            return f'&lt;img src="{result}" style="max-width:100%;border-radius:8px"/&gt;'
   125	        if result.startswith('http://') or result.startswith('https://'):
   126	            return f'&lt;img src="{result}" style="max-width:100%;border-radius:8px"/&gt;'
   127	        _log.warning(f"[ECHARTS] str result not SVG/image path/URL: {result[:200]}")
   128	
   129	    # tuple 形式 (content, artifact)
   130	    if isinstance(result, tuple) and len(result) == 2:
   131	        cnt, artifact = result
   132	        if isinstance(cnt, str):
   133	            if "&lt;svg" in cnt:
   134	                r = _svg_to_data_url(cnt)
   135	                _log.info(f"[ECHARTS] tuple SVG → iframe[:100]={r[:100]}")
   136	                return (r, artifact)
   137	            if cnt.lower().endswith(('.png', '.jpg', '.jpeg', '.gif', '.webp')):
   138	                moved = _move_echarts_image_to_workspace(cnt)
   139	                return (f'&lt;img src="file:///{moved.replace(chr(92), "/")}" style="max-width:100%;border-radius:8px"/&gt;', artifact)
   140	            if cnt.startswith('data:image'):
   141	                return (f'&lt;img src="{cnt}" style="max-width:100%;border-radius:8px"/&gt;', artifact)
   142	            if cnt.startswith('http://') or cnt.startswith('https://'):
   143	                return (f'&lt;img src="{cnt}" style="max-width:100%;border-radius:8px"/&gt;', artifact)
   144	
   145	    return result

   276	def _pack_semiotic_props(kwargs: dict) -&gt; dict:
   277	    """semiotic-mcp 的 renderChart / diagnoseConfig 等工具要求 chart 属性
   278	    （data, categoryAccessor, valueAccessor …）全部嵌套在 ``props`` 字段内：
   279	
   280	        { component: "BarChart", props: { data: [...], ... }, format: "svg" }
   281	
   282	    但 LLM 经常把所有字段平铺到顶层。本函数把非元数据字段打包进 ``props``。
   283	    同时自动修正 xy / ordinal 图表的 accessor 命名差异：
   284	        xy 图表（LineChart 等）使用 xAccessor / yAccessor
   285	        ordinal 图表（BarChart 等）使用 categoryAccessor / valueAccessor
   286	    """
   287	    # semiotic-mcp 元数据字段 + LangChain 框架内部字段，均不应被打包进 props
   288	    TOP_LEVEL_KEYS = {
   289	        "component", "props", "theme", "format", "usageMode", "viewportWidth",
   290	        # LangChain BaseTool / StructuredTool 框架参数
   291	        "config", "callbacks", "tags", "metadata", "run_name",
   292	    }
   293	    # xy 类图表使用 xAccessor/yAccessor；ordinal 类使用 categoryAccessor/valueAccessor
   294	    _XY_COMPONENTS = {
   295	        "LineChart", "AreaChart", "StackedAreaChart", "Scatterplot", "BubbleChart",
   296	        "Heatmap", "ConnectedScatterplot", "QuadrantChart", "MultiAxisLineChart",
   297	        "CandlestickChart", "DifferenceChart", "BumpChart",
   298	    }
   299	    _ORDINAL_COMPONENTS = {
   300	        "BarChart", "StackedBarChart", "GroupedBarChart", "SwarmPlot", "BoxPlot",
   301	        "DotPlot", "Histogram", "ViolinPlot", "RidgelinePlot", "PieChart",
   302	        "DonutChart", "FunnelChart", "LikertChart", "SwimlaneChart", "Treemap",
   303	        "ParallelCoordinatesChart", "SummaryChart",
   304	    }
   305	    if "component" not in kwargs:
   306	        return kwargs
   307	    if "props" in kwargs and isinstance(kwargs["props"], dict) and not any(
   308	        k not in TOP_LEVEL_KEYS for k in kwargs
   309	    ):
   310	        # props 已存在且无多余顶层字段，仍需检查 accessor 命名
   311	        pass
   312	    else:
   313	        props = kwargs.get("props") if isinstance(kwargs.get("props"), dict) else {}
   314	        extra = {k: v for k, v in kwargs.items() if k not in TOP_LEVEL_KEYS}
   315	        if not extra:
   316	            return kwargs
   317	        new_props = {**extra, **props}  # props 内已有字段优先（防止被覆盖）
   318	        kwargs = {k: v for k, v in kwargs.items() if k in TOP_LEVEL_KEYS} | {"props": new_props}
   319	
   320	    # 修正 accessor 命名
   321	    component = kwargs.get("component", "")
   322	    props = kwargs.get("props")
   323	    if isinstance(props, dict):
   324	        if component in _XY_COMPONENTS:
   325	            if "categoryAccessor" in props and "xAccessor" not in props:
   326	                props["xAccessor"] = props.pop("categoryAccessor")
   327	            if "valueAccessor" in props and "yAccessor" not in props:
   328	                props["yAccessor"] = props.pop("valueAccessor")
   329	            # LineChart 的 xScaleType 只接受 linear/log/time，不支持 ordinal。
   330	            # 当 x 值是字符串时：把分类值映射为数值索引（0,1,2…），保留原始标签
   331	            # 存到 _x_labels 供坐标轴显示。
   332	            _x_acc = props.get("xAccessor")
   333	            _data = props.get("data")
   334	            if (
   335	                component == "LineChart"
   336	                and _x_acc
   337	                and isinstance(_data, list)
   338	                and _data
   339	                and isinstance(_data[0], dict)
   340	                and isinstance(_data[0].get(_x_acc), str)
   341	            ):
   342	                # 按出现顺序建立 分类值 → 索引 映射
   343	                _seen = {}
   344	                _labels = []
   345	                for row in _data:
   346	                    v = row.get(_x_acc)
   347	                    if v not in _seen:
   348	                        _seen[v] = len(_labels)
   349	                        _labels.append(v)
   350	                # 把原始标签存到 _label 字段，供 tooltip 显示
   351	                for row in _data:
   352	                    row["_label"] = row[_x_acc]
   353	                    row[_x_acc] = _seen[row[_x_acc]]
   354	                # 确保数据点可见，方便 hover 显示 tooltip
   355	                props.setdefault("showPoints", True)
   356	                _log.warning(
   357	                    f"[_pack_chart_props] LineChart categorical x mapped to indices: "
   358	                    f"labels={_labels}")
   359	        elif component in _ORDINAL_COMPONENTS:
   360	            if "xAccessor" in props and "categoryAccessor" not in props:
   361	                props["categoryAccessor"] = props.pop("xAccessor")
   362	            if "yAccessor" in props and "valueAccessor" not in props:
   363	                props["valueAccessor"] = props.pop("yAccessor")
   364	    return kwargs

   369

   372	    original tool logic runs.

   376	    message so the NL2SQL pipeline does not crash.

   381	    original_run = getattr(tool, "_run", None)

   397	                    return (_CHART_ERROR_MSG, None)
   398	                raise

   402

   404	    if original_arun is not None:
   405	        @wraps(original_arun)

   407	            import logging

   414	            new_kwargs = _pack_chart_props(new_kwargs)

   416	                f"[_arun] {tool.name} FINAL: component={new_kwargs.get('component')}, "
   417	                f"props_keys={list(new_kwargs.get('props', {}).keys()) if isinstance(new_kwargs.get('props'), dict) else 'N/A'}, "
   418	                f"top_keys={list(new_kwargs.keys())}")
   419	            if is_chart and isinstance(new_kwargs.get("props"), dict):
   420	                _p = new_kwargs["props"]
   421	                _data = _p.get("data", [])
   422	                _first_keys = list(_data[0].keys()) if _data and isinstance(_data[0], dict) else "N/A"
   423	            try:
   424	                return _sanitize_chart_result(await original_arun(*new_args, **new_kwargs), is_chart)
   425	            except Exception as e:
   426	                _log.warning(f"[CHART] TOOL ERROR: {type(e).__name__}: {e}", exc_info=True)
   427	                if is_chart and ('ToolException' in type(e).__name__ or 'McpError' in type(e).__name__):
   428	                    return (_CHART_ERROR_MSG, None)
