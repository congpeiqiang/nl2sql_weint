# NL2SQL 后端架构优化实施手册

> **For agentic workers:** REQUIRED SUB-SKILL: Use superpowers:subagent-driven-development (recommended) or superpowers:executing-plans to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** 系统性优化 NL2SQL 后端的可扩展性、可维护性、可靠性和性能，消除 16 个架构级技术债。

**Architecture:** 分 5 个阶段递进实施：Phase 0 建立测试安全网 → Phase 1 止血（低风险快速修复）→ Phase 2 拆骨（模块拆分与集中化）→ Phase 3 筑基（架构级改进）→ Phase 4 扩展（按需推进）。每个阶段产出可独立运行的代码。

**Tech Stack:** Python 3.13, Starlette (ASGI), LangGraph, pytest, pytest-asyncio, Pydantic v2, asyncio, threading

**Spec:** [后端架构全面评估-2026-10-14.md](后端架构全面评估-2026-10-14.md)

## Global Constraints

- Python ≥ 3.13，使用 `uv` 管理依赖
- 所有新模块必须有 `__init__.py`，导出明确接口
- 中文注释和日志消息保持不变
- `noqa: BLE001` 仅在兜底 `except Exception as e:` 处使用，且必须配合 `_logger.exception()`
- 不改变现有 API 接口的请求/响应格式
- 不改变 LangGraph 原生路由的行为
- `.env.dev` 不提交 Git

## Review Focus

1. **中间件顺序破坏** — 23 层中间件有部分序约束，重构过程中任何顺序变更都可能导致静默行为错误
2. **守护线程停机丢失** — sync 线程和 eval worker 在进程重启时丢失进度，治理时不能使情况更差
3. **SQLite 并发互踩** — auth SQLite 共享连接 + 全局锁，修改时不能引入死锁
4. **SSE 流断裂** — 中间件改 ASGI 纯管道不能缓冲响应体，否则 `/runs/stream` SSE 中断
5. **内部 HTTP 自调用身份判定** — Docker IP + 无 Cookie 判 internal 的逻辑脆弱，改直连时不能丢失鉴权

"""验证中间件链注册顺序——任何顺序变更必须显式修改此测试。"""
import pytest

class TestMainAgentMiddlewareChain:
    """主 Agent 中间件链顺序约束。"""

    def _get_middleware_types(self) -> list[str]:
        """提取主 Agent 已注册的中间件类型名列表（外→内）。"""
        from agent.main_agent import agent
        # agent.middleware 是中间件实例列表
        return [type(mw).__name__ for mw in agent.middleware]

    def test_chain_length(self):
        """主 Agent 应有 17 层中间件。"""
        chain = self._get_middleware_types()
        assert len(chain) == 17, f"期望 17 层，实际 {len(chain)}: {chain}"

    def test_guard_middlewares_outermost(self):
        """门禁类中间件必须在最外层（列表头部）。"""
        chain = self._get_middleware_types()
        assert chain[0] == "ModelRequiredMiddleware"
        assert chain[1] == "QuotaErrorMiddleware"
        assert chain[2] == "ModelTimeoutMiddleware"

    def test_vfs_path_resolver_innermost(self):
        """VfsPathResolver 必须最内层（列表尾部）。"""
        chain = self._get_middleware_types()
        assert chain[-1] == "VfsPathResolverMiddleware"

    def test_message_slimmer_before_knowledge_trim(self):
        """MessageSlimmer 必须在 KnowledgeTrim 外层（index 更小）。"""
        chain = self._get_middleware_types()
        slim_idx = chain.index("MessageSlimmerMiddleware")
        trim_idx = chain.index("KnowledgeTrimMiddleware")
        assert slim_idx < trim_idx, f"Slimmer@{slim_idx} 应在 Trim@{trim_idx} 外层"


class TestNl2sqlAgentMiddlewareChain:
    """NL2SQL 子 Agent 中间件链顺序约束。"""

    def _get_middleware_types(self) -> list[str]:
        from agent.graphs.nl2sql_agent import agent
        return [type(mw).__name__ for mw in agent.middleware]

    def test_chain_length(self):
        """子 Agent 应有 22-23 层中间件（sql_approval 可选）。"""
        chain = self._get_middleware_types()
        assert 22 <= len(chain) <= 23, f"期望 22-23 层，实际 {len(chain)}: {chain}"

    def test_dynamic_mcp_outermost(self):
        """DynamicMCPTools 必须最外层。"""
        chain = self._get_middleware_types()
        assert chain[0] == "ModelRequiredMiddleware"

    def test_write_todos_last(self):
        """WriteTodosProtocol 必须最后追加。"""
        chain = self._get_middleware_types()
        assert chain[-1] == "WriteTodosProtocolMiddleware"

    def test_failure_circuit_between_slimmer_and_langfuse(self):
        """FailureCircuit 必须在 MessageSlimmer 内层、LangfuseSpan 外层。"""
        chain = self._get_middleware_types()
        fc_idx = chain.index("FailureCircuitMiddleware")
        slim_idx = chain.index("MessageSlimmerMiddleware")
        lf_idx = chain.index("LangfuseSpanMiddleware")
        assert slim_idx < fc_idx < lf_idx

```python
"""声明式中间件管道——用 Layer 枚举 + 拓扑排序替代手动排列。"""
from enum import Enum, auto
from dataclasses import dataclass, field
from typing import Any
import logging

_logger = logging.getLogger(__name__)


class Layer(Enum):
    """中间件执行阶段——从外到内。"""
    GUARD = auto()           # 门禁：ModelRequired, QuotaError, ModelTimeout
    TOOL_REGISTRY = auto()   # 工具注入：DynamicMCPTools, ToolFilter
    OBSERVABILITY = auto()   # 观测：TraceRecorder, LangfuseSpan, TokenMeter
    PROGRESS = auto()        # 进度：ProgressTracker, ProgressBoundary, WriteTodos
    CONTEXT = auto()         # 上下文：ThinkingToggle, DbContext, QueryKeywords
    SAFETY = auto()          # 安全：FilesystemThreadGuard, ExecuteGuard, SQLApproval
    OPTIMIZATION = auto()    # 优化：MessageSlimmer, KnowledgeTrim, QueryResultOffload
    GATE = auto()            # 业务门控：CaliberGate, QueryGate, CubeRouteGate, FailureCircuit
    PROMPT = auto()          # 提示词：dynamic_prompt, SkillsMiddleware


@dataclass
class MiddlewareSpec:
    layer: Layer
    name: str
    middleware: Any
    after: list[str] = field(default_factory=list)  # 同层内：必须在这些之后
    before: list[str] = field(default_factory=list) # 同层内：必须在这些之前


def build_chain(specs: list[MiddlewareSpec]) -> list:
    """拓扑排序 → 平坦中间件列表。"""
    # 按 layer 分组，层内按 after/before 约束拓扑排序
    from collections import defaultdict
    by_layer: dict[Layer, list[MiddlewareSpec]] = defaultdict(list)
    for spec in specs:
        by_layer[spec.layer].append(spec)

    result = []
    for layer in Layer:  # 按枚举顺序（外→内）
        layer_specs = by_layer.get(layer, [])
        sorted_specs = _topo_sort(layer_specs)
        result.extend(s.middleware for s in sorted_specs)

    _logger.info("[pipeline] 构建 %d 层中间件链", len(result))
    return result


def _topo_sort(specs: list[MiddlewareSpec]) -> list[MiddlewareSpec]:
    """同层内拓扑排序。检测循环依赖。"""
    name_to_spec = {s.name: s for s in specs}
    in_degree = {s.name: 0 for s in specs}
    adj = {s.name: [] for s in specs}

    for s in specs:
        for dep in s.after:
            if dep in adj:
                adj[dep].append(s.name)
                in_degree[s.name] += 1
        for dep in s.before:
            if dep in adj:
                adj[s.name].append(dep)
                in_degree[dep] += 1

    queue = [name for name, deg in in_degree.items() if deg == 0]
    order = []
    while queue:
        name = queue.pop(0)
        order.append(name)
        for neighbor in adj[name]:
            in_degree[neighbor] -= 1
            if in_degree[neighbor] == 0:
                queue.append(neighbor)

    if len(order) != len(specs):
        raise ValueError(f"中间件循环依赖检测：{set(s.name for s in specs) - set(order)}")

    return [name_to_spec[name] for name in order]
```

"""MCP 子进程连接池 — LRU 淘汰 + 懒加载。"""
import asyncio
import logging
from collections import OrderedDict
from typing import Optional

_logger = logging.getLogger(__name__)


class MCPConnectionPool:
    """per-db MCP 连接池。

    - max_active: 最大同时活跃连接数
    - 超出的连接按 LRU 淘汰
    - 首次访问时懒加载
    """

    def __init__(self, max_active: int = 5):
        self._active: OrderedDict[str, "MCPSession"] = OrderedDict()
        self._max_active = max_active
        self._lock = asyncio.Lock()

    async def acquire(self, db_name: str) -> "MCPSession":
        async with self._lock:
            if db_name in self._active:
                self._active.move_to_end(db_name)
                return self._active[db_name]

            if len(self._active) >= self._max_active:
                evicted_name, session = self._active.popitem(last=False)
                _logger.info("[mcp_pool] 淘汰 %s（LRU）", evicted_name)
                await session.close()

            session = await self._connect(db_name)
            self._active[db_name] = session
            return session

    async def _connect(self, db_name: str) -> "MCPSession":
        """创建 MCP 子进程连接。"""
        # 复用现有 mcp_tool.py 的连接逻辑
        from agent.tools.mcp_tool import _connect_server
        return await _connect_server(db_name)

    async def close_all(self):
        async with self._lock:
            for name, session in self._active.items():
                _logger.info("[mcp_pool] 关闭 %s", name)
                await session.close()
            self._active.clear()

    @property
    def active_count(self) -> int:
        return len(self._active)
```

async def _adaptive_poll(self, sub_thread_id: str):
    """自适应轮询：变化时快、稳定时慢。"""
    interval = 0.2       # 初始 200ms
    stable_count = 0
    last_todos_hash = None

    while not self._done:
        state = await self._fetch_subagent_state(sub_thread_id)
        current_hash = _hash_todos(state)

        if current_hash != last_todos_hash:
            # 状态变化 → 快速轮询
            interval = 0.2
            stable_count = 0
            last_todos_hash = current_hash
            await self._sync_to_main(state)
        else:
            # 稳定 → 逐步降频
            stable_count += 1
            interval = min(0.2 + stable_count * 0.1, 2.0)

        # 终态检测（无论是否变化都检查）
        if self._is_terminal(state):
            await self._handle_terminal(state)
            break

        await asyncio.sleep(interval)
```

"""内部 LangGraph 直连客户端——绕过 HTTP + 中间件链。

用于 API handler 内部调用同进程的 LangGraph API，
避免 HTTP 序列化/反序列化 + 完整中间件链开销。
"""
import logging

_logger = logging.getLogger(__name__)


class InternalLangGraphClient:
    """直连 checkpointer，跳过 HTTP 层。"""

    def __init__(self):
        self._checkpointer = None

    async def _ensure_checkpointer(self):
        if self._checkpointer is None:
            from agent.checkpoint.checkpointer_factory import checkpointer
            self._checkpointer = checkpointer

    async def get_state(self, thread_id: str) -> dict:
        """直接读 checkpoint，不走 HTTP。"""
        await self._ensure_checkpointer()
        config = {"configurable": {"thread_id": thread_id}}
        return await self._checkpointer.aget(config)

    async def get_history(self, thread_id: str, limit: int = 10) -> list:
        await self._ensure_checkpointer()
        config = {"configurable": {"thread_id": thread_id}}
        return await self._checkpointer.alist(config, limit=limit)

    async def update_state(self, thread_id: str, values: dict,
                           as_node: str = "__start__") -> None:
        await self._ensure_checkpointer()
        config = {
            "configurable": {
                "thread_id": thread_id,
            },
        }
        await self._checkpointer.aput(config, values)
```

"""轻量级依赖注入容器。"""
class Container:
    def __init__(self):
        self._singletons: dict[str, Any] = {}
        self._factories: dict[str, callable] = {}

    def register_singleton(self, name: str, instance: Any):
        self._singletons[name] = instance

    def register_factory(self, name: str, factory: callable):
        self._factories[name] = factory

    def resolve(self, name: str) -> Any:
        if name in self._singletons:
            return self._singletons[name]
        if name in self._factories:
            instance = self._factories[name]()
            self._singletons[name] = instance  # 缓存
            return instance
        raise KeyError(f"未注册: {name}")

# 全局容器
_container = Container()

def get_container() -> Container:
    return _container
```
